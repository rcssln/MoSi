  1  # terminal_model.py - SimPy model of one jeepney terminal (one loading bay, one waiting line).
  2  # Passengers arrive at random and join a single first-come, first-served line. Jeepneys
  3  # queue for the one loading bay, load passengers until the dispatch rule says to depart,
  4  # make a random round trip, and rejoin the queue. All times are in minutes. Statistics
  5  # are recorded only after the warm-up period.
  6  
  7  import numpy as np
  8  import simpy
  9  
 10  # Stores the time a passenger joined the line and an event that is triggered when a
 11  # jeepney takes the passenger from the line.
 12  class Passenger:
 13      def __init__(self, env, arrival_time):
 14          self.arrival_time = arrival_time
 15          self.boarded = env.event()
 16  
 17  # Holds the whole system: the passenger line, the loading bay, the random samplers,
 18  # and every counter used for the results. A test can replace any sampler with fixed
 19  # values. The whole-run counters (arrived, left_line) are used for the conservation check;
 20  # the lists and sums below them are recorded after the warm-up only.
 21  class Terminal:
 22      def __init__(self, env, params, rng, log=False):
 23          self.env = env
 24          self.p = params
 25          self.rng = rng
 26          self.line = simpy.Store(env)
 27          self.bay = simpy.Resource(env, capacity=1)
 28          self.warmup = params["warmup"]
 29  
 30          self.next_gap = params.get("arrival_sampler") or (
 31              lambda: rng.exponential(params["mean_interarrival"]))
 32          self.next_board = params.get("board_sampler") or (
 33              lambda: max(params["board_min_sec"],
 34                          rng.normal(params["board_mean_sec"], params["board_sd_sec"])) / 60)
 35          self.next_trip = params.get("trip_sampler") or (
 36              lambda: max(params["trip_min"],
 37                          rng.normal(params["trip_mean"], params["trip_sd"])))
 38  
 39          self.arrived = 0
 40          self.left_line = 0
 41          self.waits = []
 42          self.depart_waits = []
 43          self.seats = []
 44          self.idles = []
 45          self.area = 0.0
 46          self.last_t = 0.0
 47          self.bay_busy = 0.0
 48          self.bay_since = None
 49          self.log_lines = [] if log else None
 50  
 51      # Adds one line "time  event" to the event log. Does nothing when logging is off.
 52      def log(self, text):
 53          if self.log_lines is not None:
 54              self.log_lines.append(f"{self.env.now:8.2f}  {text}")
 55  
 56      # Keeps the running total of (line length x time) used to compute the average number
 57      # of passengers waiting. Must be called just before the line length changes. Time
 58      # inside the warm-up period is not counted.
 59      def track(self):
 60          now = self.env.now
 61          start = max(self.last_t, self.warmup)
 62          if now > start:
 63              self.area += len(self.line.items) * (now - start)
 64          self.last_t = now
 65  
 66      # Runs for the whole simulation. After each random inter-arrival time (Exponential),
 67      # it creates one new passenger process.
 68      def passenger_generator(self):
 69          while True:
 70              yield self.env.timeout(self.next_gap())
 71              self.env.process(self.passenger())
 72  
 73      # One passenger. Joins the back of the waiting line, then stays until load() takes
 74      # them from the front of the line.
 75      def passenger(self):
 76          pax = Passenger(self.env, self.env.now)
 77          self.arrived += 1
 78          self.track()
 79          self.line.put(pax)
 80          self.log(f"passenger arrives (line = {len(self.line.items)})")
 81          yield pax.boarded
 82  
 83      # Loads the jeepney that is in the bay and returns the passengers on board. Each
 84      # passenger takes a random boarding time. Departure rules:
 85      # baseline - depart only when all seats are full
 86      # timed    - depart when full, or when the time limit has passed since the jeepney
 87      # entered the bay; a jeepney with nobody on board keeps waiting for its
 88      # first passenger instead of leaving empty
 89      def load(self):
 90          env, cap = self.env, self.p["capacity"]
 91          timed = self.p["rule"] == "timed"
 92          deadline = env.now + self.p["time_limit"] if timed else None
 93          onboard = []
 94          while len(onboard) < cap:
 95              if timed and env.now >= deadline and len(onboard) >= 1:
 96                  break
 97              self.track()
 98              get = self.line.get()
 99              if timed and env.now < deadline:
100                  yield get | env.timeout(deadline - env.now)
101                  if not get.triggered:
102                      get.cancel()
103                      continue
104              else:
105                  yield get
106              pax = get.value
107              self.left_line += 1
108              if env.now >= self.warmup:
109                  self.waits.append(env.now - pax.arrival_time)
110              pax.boarded.succeed()
111              onboard.append(pax)
112              self.log(f"passenger boards (seat {len(onboard)}/{cap})")
113              yield env.timeout(self.next_board())
114          return onboard
115  
116      # Life cycle of one jeepney, repeated forever: join the jeepney queue, enter the bay
117      # when it is free, load, depart, travel for a random round-trip time, then rejoin the
118      # queue. At each departure after the warm-up it records the seats filled, the time
119      # spent at the terminal (including waiting for the bay), and each passenger's time
120      # from arrival to departure.
121      def jeepney(self, jid):
122          env = self.env
123          while True:
124              arrive = env.now
125              self.log(f"jeepney {jid} arrives at terminal")
126              with self.bay.request() as req:
127                  yield req
128                  bay_enter = env.now
129                  self.bay_since = bay_enter
130                  self.log(f"jeepney {jid} enters bay")
131                  onboard = yield from self.load()
132                  seats = len(onboard)
133                  self.bay_busy += max(0.0, env.now - max(bay_enter, self.warmup))
134                  self.bay_since = None
135                  if env.now >= self.warmup:
136                      self.seats.append(seats)
137                      self.idles.append(env.now - arrive)
138                      self.depart_waits += [env.now - pax.arrival_time for pax in onboard]
139                  self.log(f"jeepney {jid} departs with {seats} passengers")
140              yield env.timeout(self.next_trip())
141  
142      # Converts the recorded values into the performance measures of one replication:
143      # average and maximum wait in line, average and maximum time from arrival to
144      # departure, average passengers waiting, jeepney idle time per trip, bay utilization
145      # (fraction of time occupied), passengers served per hour, and seats filled per
146      # departure. Measures with no data are returned as nan. The whole-run counts are
147      # included so the conservation check can be made.
148      def results(self, end):
149          self.track()
150          if self.bay_since is not None:
151              self.bay_busy += end - max(self.bay_since, self.warmup)
152          measured = end - self.warmup
153          mean = lambda x: float(np.mean(x)) if len(x) else float("nan")
154          out = {
155              "avg_wait": mean(self.waits),
156              "max_wait": max(self.waits) if self.waits else float("nan"),
157              "avg_depart_wait": mean(self.depart_waits),
158              "max_depart_wait": max(self.depart_waits) if self.depart_waits else float("nan"),
159              "avg_waiting": self.area / measured,
160              "avg_idle": mean(self.idles),
161              "bay_util": self.bay_busy / measured,
162              "served_per_hour": len(self.waits) / (measured / 60),
163              "avg_seats": mean(self.seats),
164              "departures": len(self.seats),
165              "arrived": self.arrived,
166              "boarded": self.left_line,
167              "still_waiting": len(self.line.items),
168          }
169          if self.log_lines is not None:
170              out["log"] = self.log_lines
171          return out
172  
173  # Runs one replication: builds the terminal, starts the passenger arrivals and all
174  # jeepneys (every jeepney starts in the queue at time 0), runs for the warm-up plus the
175  # measured period, and returns the performance measures. The same seed always gives the
176  # same result.
177  def run_simulation(params, seed, log=False):
178      env = simpy.Environment()
179      rng = np.random.default_rng(seed)
180      term = Terminal(env, params, rng, log)
181      env.process(term.passenger_generator())
182      for jid in range(params["fleet_size"]):
183          env.process(term.jeepney(jid))
184      end = params["warmup"] + params["run_length"]
185      env.run(until=end)
186      return term.results(end)

  1  """run_experiments.py - run every scenario for several independent replications."""
  2  import os
  3  import pandas as pd
  4  import config
  5  from terminal_model import run_simulation
  6  
  7  
  8  def run_all(replications=config.REPLICATIONS):
  9      rows = []
 10      for sc in config.SCENARIOS:
 11          params = {**config.BASE_PARAMS, **sc}
 12          for rep in range(replications):
 13              # Same seed for the same replication number in every scenario, so
 14              # scenarios see similar random numbers (helps the paired comparison).
 15              seed = config.SEED_BASE + rep
 16              res = run_simulation(params, seed)
 17              rows.append({"scenario": sc["name"], "period": sc["period"],
 18                           "fleet_size": sc["fleet_size"], "rule": sc["rule"],
 19                           "time_limit": sc["time_limit"], "replication": rep,
 20                           "seed": seed, **res})
 21          print(f"done {sc['name']}")
 22      return pd.DataFrame(rows)
 23  
 24  
 25  if __name__ == "__main__":
 26      os.makedirs(config.OUTPUT_DIR, exist_ok=True)
 27      df = run_all()
 28      path = os.path.join(config.OUTPUT_DIR, "results.csv")
 29      df.to_csv(path, index=False)
 30      print(f"saved {len(df)} rows to {path}")

  1  """analyze.py - confidence intervals, comparison with baseline, charts, validation."""
  2  import os
  3  import numpy as np
  4  import pandas as pd
  5  import matplotlib
  6  matplotlib.use("Agg")
  7  import matplotlib.pyplot as plt
  8  from scipy import stats
  9  import config
 10  
 11  MEASURES = {   # column -> label with units
 12      "avg_wait": "Average passenger wait in line (min)",
 13      "max_wait": "Maximum passenger wait (min)",
 14      "avg_depart_wait": "Average time from arrival to departure (min)",
 15      "max_depart_wait": "Maximum time from arrival to departure (min)",
 16      "avg_waiting": "Average passengers waiting (people)",
 17      "avg_idle": "Jeepney time at terminal per trip (min)",
 18      "bay_util": "Bay utilization (fraction)",
 19      "served_per_hour": "Passengers served (per hour)",
 20      "avg_seats": "Seats filled per departure (people)",
 21  }
 22  OUT = config.OUTPUT_DIR
 23  
 24  
 25  def ci(x, level=0.95):
 26      """Mean and half-width of the t confidence interval."""
 27      x = np.asarray(x, dtype=float)
 28      x = x[~np.isnan(x)]
 29      n = len(x)
 30      half = stats.t.ppf((1 + level) / 2, n - 1) * x.std(ddof=1) / np.sqrt(n)
 31      return x.mean(), half
 32  
 33  
 34  def summarize(df):
 35      rows = []
 36      for name, g in df.groupby("scenario", sort=False):
 37          row = {"scenario": name, **{k: g[k].iloc[0] for k in
 38                 ("period", "fleet_size", "rule", "time_limit")}}
 39          for m in MEASURES:
 40              row[m + "_mean"], row[m + "_hw"] = ci(g[m])
 41          rows.append(row)
 42      return pd.DataFrame(rows)
 43  
 44  
 45  def compare_to_baseline(df):
 46      """Paired difference (timed - baseline) per replication, with 95% CI."""
 47      rows = []
 48      for _, sc in df.drop_duplicates("scenario").iterrows():
 49          if sc["rule"] == "baseline":
 50              continue
 51          base_name = f"{sc['period']}_N{sc['fleet_size']}_baseline"
 52          a = df[df.scenario == sc["scenario"]].set_index("replication")
 53          b = df[df.scenario == base_name].set_index("replication")
 54          row = {"scenario": sc["scenario"], "baseline": base_name}
 55          for m in MEASURES:
 56              mean, half = ci(a[m] - b[m])
 57              row[m + "_diff"], row[m + "_hw"] = mean, half
 58              row[m + "_signif"] = abs(mean) > half      # CI excludes 0
 59          rows.append(row)
 60      return pd.DataFrame(rows)
 61  
 62  
 63  def plot_vs_T(summ, measure, fname):
 64      """Measure vs time limit T, one line per (period, N); dashed = baseline."""
 65      fig, ax = plt.subplots(figsize=(7, 4.5))
 66      for (period, n), g in summ.groupby(["period", "fleet_size"]):
 67          t = g[g.rule == "timed"].sort_values("time_limit")
 68          line = ax.errorbar(t.time_limit, t[measure + "_mean"], yerr=t[measure + "_hw"],
 69                             marker="o", capsize=3, label=f"{period}, N={n}")
 70          base = g[g.rule == "baseline"][measure + "_mean"].iloc[0]
 71          ax.axhline(base, ls="--", lw=1, color=line[0].get_color())
 72      ax.set_xlabel("Time limit T (min)")
 73      ax.set_ylabel(MEASURES[measure])
 74      ax.set_title(f"{MEASURES[measure]} vs. time limit T\n(dashed = baseline, bars = 95% CI)")
 75      ax.set_xticks(config.TIME_LIMITS)
 76      ax.legend()
 77      fig.tight_layout()
 78      fig.savefig(os.path.join(OUT, fname), dpi=150)
 79      plt.close(fig)
 80  
 81  
 82  def plot_scenarios(summ, measure, fname):
 83      """Bar chart comparing all scenarios on one measure."""
 84      fig, ax = plt.subplots(figsize=(8, 6))
 85      ax.barh(summ.scenario, summ[measure + "_mean"], xerr=summ[measure + "_hw"], capsize=3)
 86      ax.set_xlabel(MEASURES[measure])
 87      ax.set_ylabel("Scenario")
 88      ax.set_title(f"{MEASURES[measure]} by scenario (95% CI)")
 89      ax.invert_yaxis()
 90      fig.tight_layout()
 91      fig.savefig(os.path.join(OUT, fname), dpi=150)
 92      plt.close(fig)
 93  
 94  
 95  def validate(scenario, observed_avg_wait, observed_avg_seats, results_csv=None):
 96      """Is each observed value inside the simulated 95% CI? Returns a dict of results."""
 97      df = pd.read_csv(results_csv or os.path.join(OUT, "results.csv"))
 98      g = df[df.scenario == scenario]
 99      out = {}
100      for label, col, obs in (("avg wait (min)", "avg_wait", observed_avg_wait),
101                              ("seats per departure", "avg_seats", observed_avg_seats)):
102          if obs is None:
103              continue
104          mean, half = ci(g[col])
105          inside = mean - half <= obs <= mean + half
106          print(f"{scenario} {label}: observed {obs:.2f}, simulated "
107                f"{mean:.2f} [{mean - half:.2f}, {mean + half:.2f}] -> "
108                f"{'INSIDE' if inside else 'OUTSIDE'} the 95% CI")
109          out[col] = inside
110      return out
111  
112  
113  if __name__ == "__main__":
114      os.makedirs(OUT, exist_ok=True)
115      df = pd.read_csv(os.path.join(OUT, "results.csv"))
116      summ = summarize(df)
117      summ.to_csv(os.path.join(OUT, "summary.csv"), index=False)
118      diff = compare_to_baseline(df)
119      diff.to_csv(os.path.join(OUT, "diff_vs_baseline.csv"), index=False)
120      if len(config.TIME_LIMITS) > 1:       # a line chart needs more than one T value
121          plot_vs_T(summ, "avg_wait", "wait_vs_T.png")
122          plot_vs_T(summ, "avg_idle", "idle_vs_T.png")
123      plot_scenarios(summ, "avg_wait", "scenarios_avg_wait.png")
124      plot_scenarios(summ, "served_per_hour", "scenarios_served.png")
125      print(summ[["scenario", "avg_wait_mean", "avg_wait_hw", "avg_idle_mean",
126                  "bay_util_mean", "avg_seats_mean"]].round(2).to_string(index=False))
127      obs = config.OBSERVED
128      if obs["avg_wait"] is not None or obs["avg_seats"] is not None:
129          validate(obs["scenario"], obs["avg_wait"], obs["avg_seats"])
130      else:
131          print("\nValidation skipped: fill config.OBSERVED with your observed values.")

  1  """sensitivity.py - where does each dispatch rule stop working well?
  2  
  3  Sweep 1: passenger arrival rate (passengers per minute), fleet size fixed.
  4  Sweep 2: fleet size N, arrival rate fixed at the peak value.
  5  Both compare the baseline rule with the timed rule (T = first value in config.TIME_LIMITS).
  6  Uses the PEAK round-trip time from config. Results go to outputs/sensitivity_*.csv.
  7  """
  8  import os
  9  import pandas as pd
 10  import config
 11  from terminal_model import run_simulation
 12  
 13  RATES_PER_MIN = [0.5, 1, 1.5, 2, 2.5, 3, 4, 5, 6, 8]    # passengers per minute
 14  FLEET_SWEEP = [6, 8, 10, 12, 15, 18, 20]
 15  REPS = config.REPLICATIONS
 16  T = config.TIME_LIMITS[0]
 17  
 18  
 19  def one_setting(rule, fleet_size, mean_gap, reps=REPS):
 20      rows = []
 21      for rep in range(reps):
 22          p = {**config.BASE_PARAMS, "rule": rule, "time_limit": T if rule == "timed" else None,
 23               "fleet_size": fleet_size, "mean_interarrival": mean_gap,
 24               "trip_mean": config.TRIP_MEAN_MIN["peak"]}
 25          rows.append(run_simulation(p, config.SEED_BASE + rep))
 26      return pd.DataFrame(rows).mean(numeric_only=True)
 27  
 28  
 29  def sweep(values, make_args, label):
 30      out = []
 31      for v in values:
 32          for rule in ("baseline", "timed"):
 33              res = one_setting(rule, *make_args(v))
 34              out.append({label: v, "rule": rule, **res.to_dict()})
 35          print(f"done {label} = {v}")
 36      return pd.DataFrame(out)
 37  
 38  
 39  if __name__ == "__main__":
 40      os.makedirs(config.OUTPUT_DIR, exist_ok=True)
 41      n = config.FLEET_SIZES[0]
 42      peak_gap = config.ARRIVAL_MEAN_MIN["peak"]
 43      a = sweep(RATES_PER_MIN, lambda r: (n, 1 / r), "arrival_per_min")
 44      a.to_csv(os.path.join(config.OUTPUT_DIR, "sensitivity_arrival.csv"), index=False)
 45      f = sweep(FLEET_SWEEP, lambda k: (k, peak_gap), "fleet_size")
 46      f.to_csv(os.path.join(config.OUTPUT_DIR, "sensitivity_fleet.csv"), index=False)
 47      print("saved sensitivity_arrival.csv and sensitivity_fleet.csv")

  2  
  3  Usage:  python fit_inputs.py my_data.csv
  4  
  5  CSV columns: event_type, and a number in a column named timestamp OR duration.
  6    event_type        number                                   unit
  7    passenger_arrival timestamp of each arrival (inter-arrivals are computed)  min
  8    jeepney_arrival   timestamp of each arrival (inter-arrivals are computed)  min
  9    boarding          duration per passenger                    sec

 24  EVENTS = {
 25      "passenger_arrival": (True, "min", "Passenger inter-arrival time"),
 26      "jeepney_arrival": (True, "min", "Jeepney inter-arrival time"),
 27      "boarding": (False, "sec", "Boarding time per passenger"),
 28      "round_trip": (False, "min", "Round-trip time"),

 32  def load_samples(path):
 33      if not os.path.exists(path):
 34          sys.exit(f"File not found: {path}\n"
 35                   "'my_data.csv' was only an example name. Pass the path to your own data file; "
 36                   "see data_template.csv for the expected columns.")
 37      df = pd.read_csv(path)
 38      cols = [c for c in ("timestamp", "duration", "value") if c in df.columns]
 39      df["x"] = df[cols].bfill(axis=1).iloc[:, 0]
 40      samples = {}
 41      for event, (is_ts, _, _) in EVENTS.items():
 42          x = df.loc[df.event_type == event, "x"].dropna().to_numpy(float)
 43          if is_ts:
 44              x = np.diff(np.sort(x))          # timestamps -> gaps between events
 45          if len(x) >= 5:
 46              samples[event] = x
 47      return samples

 50  def fit_candidates(x):
 51      """Return {name: frozen scipy distribution}."""
 52      mean = x.mean()
 53      cands = {}      # order = plot order
 54      mu, sd = stats.norm.fit(x)
 55      cands["normal"] = stats.norm(mu, sd)
 56      if x.min() > 0:                          # Exponential/Erlang need positive data
 57          cands["exponential"] = stats.expon(loc=0, scale=mean)
 58          shape, _, _ = stats.gamma.fit(x, floc=0)
 59          k = max(1, int(round(shape)))        # Erlang = gamma with integer shape
 60          cands["erlang"] = stats.erlang(a=k, scale=mean / k)
 61      else:
 62          print("   (non-positive values found: skipping Exponential and Erlang)")
 63      return cands
 64

 66  def describe(name, dist):
 67      if name == "exponential":
 68          return {"mean": float(dist.kwds["scale"])}
 69      if name == "normal":
 70          return {"mean": float(dist.mean()), "sd": float(dist.std())}
 71      k = dist.args[0] if dist.args else dist.kwds["a"]
 72      return {"k": int(k), "scale": float(dist.kwds["scale"])}
 73  
 74  
 75  def fit_and_plot(event, x, outdir):
 76      _, unit, label = EVENTS[event]
 77      cands = fit_candidates(x)
 78      report = {"n": len(x), "sample_mean": float(x.mean()), "sample_sd": float(x.std(ddof=1)),
 79                "fits": {}}
 80      fig, axes = plt.subplots(1, 1 + len(cands), figsize=(4.5 * (1 + len(cands)), 4))
 81      axes[0].hist(x, bins="auto", density=True, alpha=0.5, label="data")
 82      grid = np.linspace(x.min(), x.max(), 200)
 83      p = (np.arange(1, len(x) + 1) - 0.5) / len(x)
 84      for ax, (name, dist) in zip(axes[1:], cands.items()):
 85          ks = stats.kstest(x, dist.cdf)       # note: p-value is optimistic for fitted params
 86          report["fits"][name] = {**describe(name, dist), "ks_stat": float(ks.statistic),
 87                                  "ks_p": float(ks.pvalue)}
 88          axes[0].plot(grid, dist.pdf(grid), label=name)
 89          ax.scatter(dist.ppf(p), np.sort(x), s=8)
 90          lim = [min(x.min(), dist.ppf(p).min()), max(x.max(), dist.ppf(p).max())]
 91          ax.plot(lim, lim, "r--")
 92          ax.set_title(f"Q-Q plot: {name} (KS p={ks.pvalue:.3f})")
 93          ax.set_xlabel(f"Theoretical quantiles ({unit})")
 94          ax.set_ylabel(f"Sample quantiles ({unit})")
 95      axes[0].set_title(f"{label}: histogram + fits")
 96      axes[0].set_xlabel(f"Time ({unit})")
 97      axes[0].set_ylabel("Density")
 98      axes[0].legend()
 99      fig.tight_layout()
100      fig.savefig(os.path.join(outdir, f"fit_{event}.png"), dpi=150)
101      plt.close(fig)
102      report["best_by_ks"] = min(report["fits"], key=lambda k: report["fits"][k]["ks_stat"])
103      return report

106  if __name__ == "__main__":
107      if len(sys.argv) < 2:
108          sys.exit("usage: python fit_inputs.py my_data.csv")
109      os.makedirs(config.OUTPUT_DIR, exist_ok=True)
110      result = {}
111      for event, x in load_samples(sys.argv[1]).items():
112          result[event] = fit_and_plot(event, x, config.OUTPUT_DIR)
113          print(f"{event}: n={len(x)}, best fit by KS = {result[event]['best_by_ks']}")

116      path = os.path.join(config.OUTPUT_DIR, "fitted_params.json")
